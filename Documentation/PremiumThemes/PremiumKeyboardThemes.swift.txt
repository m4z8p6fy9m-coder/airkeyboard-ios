import SwiftUI
import UIKit

struct KeyboardThemeAppearance {
    let keyboardBackground: Color
    let keyBackground: Color
    let specialKeyBackground: Color
    let keyText: Color
    let secondaryText: Color
    let keyBorder: Color
    let pressedOverlay: Color
    let accent: Color
}

enum KeyboardThemeSurfaceStyle {
    case solid
    case glass
}

struct KeyboardThemeDefinition: Identifiable {
    let id: String
    let name: String
    let surfaceStyle: KeyboardThemeSurfaceStyle
    let light: KeyboardThemeAppearance
    let dark: KeyboardThemeAppearance
    let followsSystemAppearance: Bool

    func appearance(for colorScheme: ColorScheme) -> KeyboardThemeAppearance {
        guard followsSystemAppearance else { return dark }
        return colorScheme == .dark ? dark : light
    }
}

private extension Color {
    /// Accepts #RRGGBB or #RRGGBBAA. Alpha is in the final two hex digits.
    init(themeHex: String) {
        let cleaned = themeHex.trimmingCharacters(in: CharacterSet.alphanumerics.inverted)
        var value: UInt64 = 0
        Scanner(string: cleaned).scanHexInt64(&value)
        let hasAlpha = cleaned.count == 8
        let r, g, b, a: Double
        if hasAlpha {
            r = Double((value >> 24) & 0xFF) / 255.0
            g = Double((value >> 16) & 0xFF) / 255.0
            b = Double((value >> 8) & 0xFF) / 255.0
            a = Double(value & 0xFF) / 255.0
        } else {
            r = Double((value >> 16) & 0xFF) / 255.0
            g = Double((value >> 8) & 0xFF) / 255.0
            b = Double(value & 0xFF) / 255.0
            a = 1.0
        }
        self.init(.sRGB, red: r, green: g, blue: b, opacity: a)
    }
}
